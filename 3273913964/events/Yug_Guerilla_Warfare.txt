country_event = {  # Victory check
	id = ygw_victory.1
	is_triggered_only = yes
	hidden = yes
	trigger = {
		OR = {
			has_country_flag = ygw_mnt
			has_country_flag = ygw_mac
			has_country_flag = ygw_bos
		}
	}
	option = {
		country_event = { 
			id = ygw_victory.1
			days = 40
		}
		### 6 Parts Yugoslavia wins
		if = {
			limit = {
				YUG = {
					controls_state = 105
					controls_state = 916
					controls_state = 909
					has_idea = YUG_federal_yug_hss
				}
				OR = {
					AND = {check_variable = { ygw_mnt_w_pow > 80 }	check_variable = { ygw_mnt_w_pop > 50 }  has_country_flag = ygw_mnt_ben_w}
					AND = {check_variable = { ygw_mnt_w_pow > 50 }	check_variable = { ygw_mnt_w_pop > 80 }  has_country_flag = ygw_mnt_ben_w}
				}
				OR = {
					AND = {check_variable = { ygw_mac_s_pow > 80 }	check_variable = { ygw_mac_s_pop > 50 }  has_country_flag = ygw_mac_ben_s}
					AND = {check_variable = { ygw_mac_s_pow > 50 }	check_variable = { ygw_mac_s_pop > 80 }  has_country_flag = ygw_mac_ben_s}
				}
				OR = {
					AND = {check_variable = { ygw_bos_s_pow > 80 }	check_variable = { ygw_bos_s_pop > 50 }  has_country_flag = ygw_bos_ben_s}
					AND = {check_variable = { ygw_bos_s_pow > 50 }	check_variable = { ygw_bos_s_pop > 80 }  has_country_flag = ygw_bos_ben_s}
				}
			}	
			country_event = { id = yugoslavia.248 }
		}
	}
}
