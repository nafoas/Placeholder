# Generic France Partition
country_event = {
	id = occupation.1
	title = occupation.1.t
	desc = occupation.1.d
	picture = GFX_event_Generic_occupation
	
	is_triggered_only = yes
	
	option = { # partition - full
		ai_chance = {
			factor = 15
		}
		name = occupation.1.a
		trigger = {
			OR = {
				controls_state = 1
				controls_state = 14
			}
			NOT = {
				country_exists = BRI
				country_exists = CRS
			}
		}

		if = {
			limit = {
				controls_state = 14
				BRI = {
					OR = {
						exists = no
						is_puppet_of = ROOT
					}
				}
			}
			add_BRI_cores = yes
			every_owned_state = {
				limit = {
					is_core_of = BRI
					is_not_claim_or_core_of_ROOT = yes

					transfer_state = PREV
				}
			}
			hidden_effect = {
				puppet = BRI
				BRI = {
					COALITION_RESET = yes
					COALITION_set_up_array = yes
				}
			}
		}

		if = {
			limit = {
				controls_state = 1
				CRS = {
					OR = {
						exists = no
						is_puppet_of = ROOT
					}
				}
			}
			add_CRS_cores = yes

				limit = {
					is_core_of = CRS
					is_not_claim_or_core_of_ROOT = yes
				}
				CRS = {
					transfer_state = PREV
				}
			}
			hidden_effect = {
				puppet = CRS
				CRS = {
					COALITION_RESET = yes
					COALITION_set_up_array = yes
				}
			}
		}

		every_owned_state = {
			limit = {
				is_core_of = FRA
				is_not_claim_or_core_of_ROOT = yes
			}
			FRA = {
				transfer_state = PREV
			}
		}

		custom_effect_tooltip = lib_countries_becomes_puppet_tt
		hidden_effect = {
			puppet = FRA
			FRA = {
				COALITION_RESET = yes
				COALITION_set_up_array = yes
			}
		}

		ai_chance = {
			factor = 75
		}
		name = occupation.1.c

				is_core_of = FRA
