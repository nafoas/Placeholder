add_namespace = can_elections

# 1935
country_event = {
	id = can_elections.1
	title = can_elections.1.t
	desc = can_elections.1.d
	picture = GFX_report_event_election_vote

	fire_only_once = yes
	is_triggered_only = yes
	trigger = {
		tag = CAN
		date > 1935.1.1
		date < 1936.1.1
	}
	
	option = { #Liberals
		name = can_elections.1.a
		ai_chance = {
			factor = 70
		}
		add_popularity = {
	   		ideology = market_liberalism
			popularity = 0.1
		}
		if = {
			limit = {
				NOT = {
					has_government = market_liberalism
				}
			}
			set_politics = { ruling_party = market_liberalism elections_allowed = yes }
			COALITION_RESET = yes
			COALITION_add_soclib = yes
			remove_ideas_with_trait = hog
			add_ideas = HoG_Mackenzie_King
			remove_ideas_with_trait = frnmin
			add_ideas = FM_Mackenzie_King
			remove_ideas_with_trait = ecomin
			add_ideas = EM_Gharles_Avery_Dunning
			remove_ideas_with_trait = secmin
			add_ideas = SM_Ernest_Lapointe
		}
	}
	
	option = { #Conservatives
		name = can_elections.1.b
		ai_chance = {
			factor = 30
			modifier = {
				is_historical_focus_on = yes
				factor = 0
			}
		}
		add_popularity = {
	   		ideology = democratic
			popularity = 0.1
		}
		if = {
			limit = {
				NOT = {
					has_government = democratic
				}
			}
			COALITION_RESET = yes
			COALITION_add_cen = yes
			remove_ideas_with_trait = hog
			add_ideas = HoG_R_B_Benett
			remove_ideas_with_trait = frnmin
			add_ideas = FM_R_B_Bennett
			remove_ideas_with_trait = ecomin
			add_ideas = EM_Edgar_Nelson_Rhodes
			remove_ideas_with_trait = secmin
			add_ideas = SM_George_Reginald_Geary
			set_politics = { ruling_party = democratic elections_allowed = yes }
		}
	}
}
