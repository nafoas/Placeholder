#Fall of Warsaw
news_event = {
	id = occ_news.1
	title = occ_news.1.t
	desc = {
		text = occ_news.1.desc.germany
		trigger = {
			event_target:new_controller = { tag = GER }
			event_target:old_controller = { tag = POL }
		}
	}
	desc = {
		text = occ_news.1.desc.generic
		trigger = {
			NOT = { event_target:new_controller = { tag = GER } }
		}
	}
	picture = GFX_event_Fall_of_Warsaw

	major = yes
	is_triggered_only = yes

	immediate = {
		10 = {
			set_state_flag = state_has_fallen
			owner = { save_event_target_as = old_controller }
			controller = { save_event_target_as = new_controller }
		}
		if = {
			limit = {
				event_target:new_controller = { tag = GER }
				NOT = { 10 = { has_state_flag = warsaw_fallen_to_GER } }
			}
			10 = { set_state_flag = warsaw_fallen_to_GER }
		}
		else_if = {
			limit = {
				event_target:new_controller = { tag = SOV }
				NOT = { 10 = { has_state_flag = warsaw_fallen_to_SOV } }
			}
			10 = { set_state_flag = warsaw_fallen_to_SOV }
		}
	}

	option = { #For old controller
		name = occ_news.1.a
		trigger = {
			tag = event_target:old_controller
		}
	}

	option = { #For Generic new controller
		name = occ_news.1.b
		trigger = {
			tag = event_target:new_controller
			NOT = {
				event_target:new_controller = { tag = SOV }
				event_target:new_controller = { original_tag = GER }
			}
		}
	}

	option = { #For Germany
		name = occ_news.1.germany
		trigger = {
			tag = event_target:new_controller
			event_target:new_controller = { tag = GER }
		}
	}

	option = { #For Soviets
		name = occ_news.1.soviet
		trigger = {
			tag = event_target:new_controller
			event_target:new_controller = { tag = SOV }
		}
	}

	option = { #Generic neutral
		name = occ_news.1.other
		trigger = {
			NOT = {
				tag = event_target:old_controller
				tag = event_target:new_controller
			}
		}
	}
}

		trigger = {
			tag = event_target:new_controller
		}

		name = occ_news.2.c

		}
