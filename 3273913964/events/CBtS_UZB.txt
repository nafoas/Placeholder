		name = uzb_internal.28.a

			base = 100
		}
		hidden_effect = {
			clr_country_flag = TKS_Fed_mechanic
		}
		if = {
			limit = {
				check_variable = {
					var = constitution_power
					value = 2
					compare = less_than_or_equals
				}
			}
			add_ideas = {
				TKS_constitution_weaker
			}
		}
		else_if = {
			limit = {
				OR = {
					check_variable = {
						var = constitution_power
						value = 3
						compare = equals
					}
					check_variable = {
						var = constitution_power
						value = 4
						compare = equals
					}
				}
			}
			add_ideas = {
				TKS_constitution_weak
			}
		}
		else_if = {
			limit = {
				OR = {
					check_variable = {
						var = constitution_power
						value = 5
						compare = equals
					}
					check_variable = {
						var = constitution_power
						value = 6
						compare = equals
					}
				}
			}
			add_ideas = {
				TKS_constitution_strong
			}
		}
		else_if = {
			limit = {
				check_variable = {
					var = constitution_power
					value = 7
					compare = greater_than_or_equals
				}
			}
			add_ideas = {
				TKS_constitution_stronger
			}
		}
	}
}
