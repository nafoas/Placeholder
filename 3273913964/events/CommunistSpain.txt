			limit = {
				check_variable = { pce_support > cnt_support }
				check_variable = { pce_support > prieto_support }
				check_variable = { pce_support > caballero_support }
			}
			country_event = { days = 5 id = spc.12 }
		}
		if = {
			limit = {
				check_variable = { cnt_support > pce_support }
				check_variable = { cnt_support > prieto_support }
				check_variable = { cnt_support > caballero_support }
			}
			country_event = { days = 5 id = spc.3 }
		}
		if = {
			limit = {
				check_variable = { caballero_support > cnt_support }
				check_variable = { caballero_support > prieto_support }
				check_variable = { caballero_support > pce_support }
			}
			country_event = { days = 5 id = spc.13 }
		}
