# Right-wing bloc supports liberals
country_event = {
	id = parliament.104
	title = parliament.104.t
	desc = parliament.104.d
	picture = GFX_Event_Spain_Cortes

	is_triggered_only = yes
	trigger = {
		NOT = { has_country_flag = parliament_closed }
	}
	fire_only_once = yes
	immediate = {
		hidden_effect = {
			country_event = { id = parliament.107 random_days = 25 }
		}
	}

	option = {
		name = parliament.104.a
		add_political_power = -25
		add_to_variable = { var = supseat value = SPR_rightist_seats }
		add_to_variable = { var = supseat value = mon_seats }
		add_to_variable = { var = supseat value = nat_seats }
		add_to_variable = { var = supseat value = fas_seats }
		set_country_flag = right_sup_opp
		if = {
			limit = {
				NOT = {
					has_country_flag = spr_op_cons
					AND = {
						is_in_array = {
							array = coalition_party_list
							value = 7
						}
						NOT = { has_country_flag = gathered_coalition }
					}
				}
			}
			add_to_variable = { var = supseat value = SPR_rightist_seats }
			set_country_flag = cons_support_opp
		}
	}
}
