country_event = {
	id = ger_civil_war.1
	title = ger_civil_war.1.t
	desc = ger_civil_war.1.d
	picture = GFX_Events_Germany_Pact_of_Steel
	is_triggered_only = yes
	option = { #continue
		name = ger_civil_war.1.a
		ai_chance = {
			base = 4
		}
		send_equipment = {
			equipment = infantry_equipment
			amount = 1200
			target = FROM
		}
		FROM = {
			country_event = {
				id = ger_civil_war.2
				days = 1
			}
			add_opinion_modifier = {
				modifier = GER_GCW_Recognized_Nationalists
				target = ROOT
			}
		}
		every_country = {
			limit = {
				tag = GRM
				has_war_with = FROM
			}
			add_opinion_modifier = {
				modifier = GER_GCW_Recognized_Nationalists_Ext
				target = ROOT
			}
		}
	}
	option = {
		name = ger_civil_war.1.b
		ai_chance = {
			base = 96
		}

	desc = ger_civil_war.11.d
	picture = GFX_event_GER_GCW_Coup

		hidden_effect = {
			add_stability = -0.20

			}

				remove_idea = GER_reactionary_plot

			}

				hugenberg

				bumke
			}
