# Spanish Civil War
news_event = {
	id = spa_news.6
	title = spa_news.6.t
	desc = spa_news.6.d
	picture = GFX_news_event_spr_spanish_civil_war
	is_triggered_only = yes
	major = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					has_global_flag = CBTS_Music_Mod_Active
				}
				SPR = {
					scoped_play_song = "a_las_barricadas"
				}
				SPA = {
					scoped_play_song = "marcha_real"
				}
				if = {
					limit = {
						ITA = {
							has_government = fascism
							exists = yes
						}
					}
					ITA = {
						scoped_play_song = "arriba_espana"
					}
				}
				if = {
					limit = {
						is_ai = no
						OR = {
							tag = USA
							tag = MEX
							tag = FRA
							tag = ENG
							tag = HOL
							tag = BEL
							tag = SOV
							tag = DEN
							tag = NOR
							tag = SWE
							tag = FIN
							tag = CZE
							tag = YUG
							tag = IRE
							tag = SWI
							tag = POL
							tag = DDR
							tag = CAN
							tag = ARG
						}
						not_has_autocratic_government = yes
					}
					scoped_play_song = "a_las_barricadas"
				}
				if = {
					limit = {
						is_ai = no
						OR = {
							tag = POR
							tag = GER
							tag = HUN
						}
						has_autocratic_government = yes
					}
					scoped_play_song = "marcha_real"
				}
			}
		}
	}

	option = {
		trigger = {
			NOT = {
				tag = SPR
				tag = SPA
			}
		}
		name = spa_news.6.a
	}

	option = {
		trigger = {
			OR = {
				tag = SPR
				tag = SPA
			}
		}
		name = spa_news.6.b
	}
}
