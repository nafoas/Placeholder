### Soviet Famine manpower loss calculation
country_event = {
	id = test.17
	title = test.17.t
	desc = test.17.desc
	is_triggered_only = yes
	hidden = yes

	option = {
		name = test.17.b
		if = {
			limit = {
				has_country_flag = soviet_famines_flag
			}
			every_owned_state = {
				limit = {
					has_dynamic_modifier = {
						modifier = Soviet_Big_Famine_modifier
					}
				}
				set_temp_variable = { current_state_pop = init_state_pop }
				multiply_temp_variable = {
					var = current_state_pop
					value = 100
				}
				multiply_temp_variable = {
					var = current_state_pop
					value = THIS.soviet_famine_modifier
				}
				#modifier@montly_population_state
			
				round_temp_variable = current_state_pop
				add_manpower = current_state_pop
			}

			country_event = {
				id = test.17
				days = 30
			}
		}
	}
}
