country_event = {
	id = mengjiang.2 # Creation of the Chanan Autonomous Government
	title = mengjiang.2.t
	desc = mengjiang.2.d
	picture = GFX_report_event_chinese_soldiers

	fire_only_once = yes 
	is_triggered_only = yes


	option = {
		name = mengjiang.2.a
		custom_effect_tooltip = "The South Chahar Autonomous Government will be created"
		hidden_effect = {
			create_dynamic_country = {
				original_tag = MEN
				save_global_event_target_as = CAG_tag
				add_state_core = 939
				transfer_state = 939
				set_cosmetic_tag = CAG
				set_capital = {
					state = 939
					remember_old_capital = no
				}
				set_technology = {
					infantry_weapons1 = 1
					generic_infantry = 1
				}
				inherit_technology = MEN
				
				reverse_add_opinion_modifier = {
					target = SOV
					modifier = trade_enemies
				}
				add_opinion_modifier = {
					target = SOV
					modifier = trade_enemies
				}
				
				set_politics = {
					ruling_party = monarchism
					last_election = "1936.1.1"
					election_frequency = 48
					elections_allowed = no
				}
				set_popularities = {
					communism = 0
					authoritarian_socialism = 0
					socialism = 0
					social_democracy = 0
					social_liberalism = 0
					market_liberalism = 0
					jacobin = 0
					democratic = 0
					neutrality = 0
					monarchism = 80
					nationalism = 0
					fascism = 20
				}
				set_research_slots = 0
			}
			JAP = {
				puppet = CAG
			}
			939 = { set_state_name = "South Chahar" }
			if = {
				limit = {
					country_exists = CAG
					country_exists = JAG
					country_exists = MEN
				}
				JAP = {
					country_event = {
						id = mengjiang.5
						days = 686
					}
				}
			}
		}
	}
}
