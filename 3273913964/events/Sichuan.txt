country_event = {
	id = sichuan.13
	title = sichuan.13.t
	desc = sichuan.13.d
	picture = GFX_report_event_chinese_soldiers_running
	is_triggered_only = yes
	hidden = yes
	option = {
		name = sichuan.13.a
		add_to_variable = { defections_counter = 1 }
		if = {
			limit = {
				check_variable = { defections_counter < 6 } #We want 5 iterations
				has_global_flag = Two_Liu_war_flag
			}
			#the amount of units transfered is defined by a fraction. We want to subtract 1/6.4 of the original amount of divisions (19) each time (19 * 1/6.4 = 2.968 which is ~ 3 divisions each time)
			#every time this runs the denominator will be lowered by 1 unit. So the fraction is 1/n-1 with n starting in 6. e: 1/6 then 1/5 then 1/4 and so on.
			set_variable = { defection_fraction_num = 1 }
			divide_variable = { defection_fraction_num = defection_fraction_den }
			random_list = {
				45 = {
					transfer_units_fraction = {
						target = LXI
						size = 0
						stockpile_ratio = 0
						army_ratio = defection_fraction_num
					}
				}
				45 = {
					transfer_units_fraction = {
						target = DXH
						size = 0
						stockpile_ratio = 0
						army_ratio = defection_fraction_num
					}
				}
				10 = {
					transfer_units_fraction = {
						target = EQG #This tag will probably never be released so its like deleting the divisions
						size = 0
						stockpile_ratio = 0
						army_ratio = defection_fraction_num
					}
				}
			}
			subtract_from_variable = { defection_fraction_den = 1 }
			country_event = {
				id = sichuan.13
				days = 30
			}
		}
		else = {
			clear_variable = defections_counter
			clear_variable = defection_fraction_num
			clear_variable = defection_fraction_den
		}
	}
}
