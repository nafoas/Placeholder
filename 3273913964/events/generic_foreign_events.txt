	desc = genf.1.d

	is_triggered_only = yes
	
	option = { #yes
		name = genf.1.a

		}
		FROM = { country_event = { id = genf.2 hours = 4 } }
	}
	
	option = { #no
		name = genf.1.b

		    base = 0

					has_country_flag = will_cause_faction_conflict
				}

			}
			modifier = {
				tag = FRA
				has_global_flag = little_entente_created
				FROM = {
					has_far_left_government = yes
					has_government = fascism
				}
				factor = 0
			}
		}
		FROM = { country_event = { id = genf.3 hours = 4 } }
	}
	
}
