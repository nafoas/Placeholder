country_event = { #1934 Midterms
	id = unitedstates.44
	title = unitedstates.44.t
	desc = unitedstates.44.d
	picture = GFX_report_event_election_vote
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		has_government = social_liberalism
		has_capitulated = no
		is_subject = no
	}
	immediate = {
		if = {
			limit = {
				has_country_flag = usa_campaign_intervention
			}
			set_variable = { var = usa_midterm_president value = usa_president_popularity }
			multiply_variable = { var = usa_midterm_president value = 0.65 }
			add_to_variable = { var = usa_democratic_pop value = party_popularity@social_liberalism }
			add_to_variable = { var = usa_democratic_pop value = party_popularity@democratic }
			set_variable = { var = usa_midterm_ideology value = usa_democratic_pop }
			multiply_variable = { var = usa_midterm_ideology value = 0.35 }
			multiply_variable = { var = usa_midterm_ideology value = 100 }
			add_to_variable = { var = usa_midterm_results value = usa_midterm_president }
			add_to_variable = { var = usa_midterm_results value = usa_midterm_ideology }
			clr_country_flag = usa_campaign_intervention
		}

		if = {
			limit = {
				has_country_flag = usa_campaign_partial
			}
			set_variable = { var = usa_midterm_president value = usa_president_popularity }
			multiply_variable = { var = usa_midterm_president value = 0.45 }
			add_to_variable = { var = usa_democratic_pop value = party_popularity@social_liberalism }
			add_to_variable = { var = usa_democratic_pop value = party_popularity@democratic }
			set_variable = { var = usa_midterm_ideology value = usa_democratic_pop }
			multiply_variable = { var = usa_midterm_ideology value = 0.55 }
			multiply_variable = { var = usa_midterm_ideology value = 100 }
			add_to_variable = { var = usa_midterm_results value = usa_midterm_president }
			add_to_variable = { var = usa_midterm_results value = usa_midterm_ideology }
			clr_country_flag = usa_campaign_partial
		}

		if = {
			limit = {
				has_country_flag = usa_campaign_no
			}
			set_variable = { var = usa_midterm_president value = usa_president_popularity }
			multiply_variable = { var = usa_midterm_president value = 0.25 }
			add_to_variable = { var = usa_democratic_pop value = party_popularity@social_liberalism }
			add_to_variable = { var = usa_democratic_pop value = party_popularity@democratic }
			set_variable = { var = usa_midterm_ideology value = usa_democratic_pop }
			multiply_variable = { var = usa_midterm_ideology value = 0.75 }
			multiply_variable = { var = usa_midterm_ideology value = 100 }
			add_to_variable = { var = usa_midterm_results value = usa_midterm_president }
			add_to_variable = { var = usa_midterm_results value = usa_midterm_ideology }
			clr_country_flag = usa_campaign_no
		}

		# Results
		if = { # >= 70
			limit = {
				check_variable = {
					var = usa_midterm_results
					value = 70
					compare = greater_than_or_equals
				}
			}
			country_event = { #Landslide Victory
				id = unitedstates.45
				hours = 12
				random_hours = 4
			}
			random_list = {
				40 = {
					country_event = {
						id = USA_flavour.13
						days = 1
					}
				}
				60 = {
					country_event = {
						id = USA_flavour.14
						days = 1
					}
				}
			}
		}
		else_if = { #50-69
			limit = {
				# check_variable = { usa_midterm_results < 70 }
				check_variable = {
					var = usa_midterm_results
					value = 50
					compare = greater_than_or_equals
				}
			}
			country_event = { #Victory
				id = unitedstates.46
				hours = 12
				random_hours = 4
			}
			random_list = {
				40 = {
					country_event = {
						id = USA_flavour.13
						days = 1
					}
				}
				60 = {
					country_event = {
						id = USA_flavour.14
						days = 1
					}
				}
			}
		}
		else_if = { #30-49
			limit = {
				# check_variable = { usa_midterm_results < 50 }
				check_variable = {
					var = usa_midterm_results
					value = 30
					compare = greater_than_or_equals
				}
			}
			country_event = { #Defeat
				id = unitedstates.47
				hours = 12
				random_hours = 4
			}
		}
		else = { # < 30
			country_event = { #Disastrous Defeat
				id = unitedstates.1000
				hours = 12
				random_hours = 4
			}
		}

		set_variable = { var = usa_midterm_results value = 0 }
		set_variable = { var = usa_democratic_pop value = 0 }
		set_variable = { var = usa_midterm_ideology value = 0 }
		set_variable = { var = usa_midterm_president value = 0 }
	}

	option = { #And the results are...
		name = unitedstates.44.a
	}
}
