		}

			ENG_is_not_voting = yes

			date < 1934.1.1
		}
		ai_will_do = {
			factor = 100
			modifier = {
				OR = {
					date < 1933.6.1
					date > 1933.8.1
				}
				factor = 0.05
			}
		}
		cost = 15
		fire_only_once = yes #Leave this for now
		complete_effect = {
			ENG_reset_supseat = yes
			ENG_set_voting_flag = yes #Hides other laws

			ENG_set_government_seats = yes
			hidden_effect = {
			}
			effect_tooltip = { #Effect of the law if approved
				add_political_power = 45
			}
		}
		days_remove = 15 #Time to vote
		remove_effect = { #Effect upon voting
			ENG_clear_voting_flag = yes #Shows other laws again
			ENG_clr_austerity_law = yes
			ENG_clr_negotiated_flags_if_necessary = yes
			hidden_effect = {
				if = {
					limit = {
						ENG_law_passed = yes
					}

					ENG_bounds_checking = yes
				}

					ENG_vote_lost = yes

			}
		}
	}
