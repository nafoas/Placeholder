	ARM_AZR_start_war = {
		icon = generic_ignite_civil_war
		allowed = {
			OR = {
				tag = ARM
				tag = AZR
			}
		}
		priority = 9
		available = {
			if = {
				limit = { tag = ARM }
				NOT = { has_border_war_with = AZR }
			}
			if = {
				limit = { tag = AZR }
				NOT = { has_border_war_with = ARM }
			}
		}
		activation = {
			always = no
		}
		is_good = no
		days_mission_timeout = 10
		# fire_only_once = yes
		timeout_effect = {
			clear_variable = global.SC_Tension
			if = {
				limit = {
					check_variable = {
						global.SC_most_recent_attacker = ARM
					}
				}
				ARM = {
					declare_war_on = {
						target = AZR
						type = take_claimed_state
					}
				}
			}
			else = {
				AZR = {
					declare_war_on = {
						target = ARM
						type = take_claimed_state
					}
				}
			}
			clear_variable = global.SC_most_recent_attacker
		}
	}
