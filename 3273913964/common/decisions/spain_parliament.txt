	SPR_italian_fund = {
		icon = generic_tank
		allowed = { tag = SPR }

		available = {
			NOT = { has_country_flag = law_passing }
			NOT = {
				OR = {
					has_idea = SPR_parliament_protectionism
					has_idea = SPR_gov_help
					has_idea = SPR_public_spending
					has_idea = SPR_worker_rights
					has_idea = SPR_church_subsidy2
				}
			}
		}
		visible = {
			NOT = { has_country_flag = law_passing }
			has_country_flag = SPR_italy_unlock
		}
		cancel_trigger = {
			hidden_trigger = {
				has_country_flag = parliament_closed
			}
		}
		cost = 0
		fire_only_once = yes

		complete_effect = {
			set_country_flag = law_passing
			custom_effect_tooltip = will_vote
			set_country_flag = law_passing_gen
			set_variable = { supseat = colseat }
			effect_tooltip = {
				add_political_power = 10
				add_war_support = 0.05
				custom_effect_tooltip = SPR_will_grant_funds
			}
		}
		days_remove = 90
		remove_effect = {
			#custom_effect_tooltip = will_vote
			hidden_effect = {
				if = {
					limit = {
						check_variable = {
							var = supseat
							value = min_sup_seat
							compare = greater_than_or_equals
						}
					}
					country_event = { id = parliament.100 }
					add_political_power = 10
					add_war_support = 0.05
					set_country_flag = SPR_parliament_ita_funds
					if = {
						limit = {
							has_country_flag = protectionism_promised
						}
						add_timed_idea = {
							idea = SPR_parliament_protectionism
							days = 150
						}
					}
					if = {
						limit = {
							has_country_flag = government_help
						}
						add_timed_idea = {
							idea = SPR_gov_help
							days = 150
						}
					}

					if = {
						limit = {
							has_country_flag = public_spending
						}
						add_timed_idea = {
							idea = SPR_public_spending
							days = 150
						}
					}

					if = {
						limit = {
							has_country_flag = promised_rights
						}
						add_timed_idea = {
							idea = SPR_worker_rights
							days = 150
						}
					}
					if = {
						limit = {
							has_country_flag = promised_church
						}
						add_timed_idea = {
							idea = SPR_church_subsidy2
							days = 150
						}
					}

				}
				else = {
					country_event = { id = parliament.1 }
				}
				set_variable = { supseat = 0 }
				clr_country_flag = law_passing
				clr_country_flag = SPR_already_neg_far_left
				clr_country_flag = already_socs
				clr_country_flag = already_socdems
				clr_country_flag = SPR_already_neg_cen_rep
				clr_country_flag = SPR_already_neg_libs
				clr_country_flag = SPR_already_neg_right
				clr_country_flag = already_aut
				clr_country_flag = SPR_already_neg_fas
				clr_country_flag = law_passing_gen
				clr_country_flag = protectionism_promised
				clr_country_flag = government_help
				clr_country_flag = public_spending
				clr_country_flag = promised_rights
				clr_country_flag = promised_church
				clr_country_flag = already_defend
			}
		}
