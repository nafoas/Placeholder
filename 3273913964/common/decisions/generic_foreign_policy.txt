	GEN_join_allies = {
		fire_only_once = yes
		icon = generic_army_support
		cosmetic_tag = ENG
		cosmetic_ideology = democratic
		allowed = {
			NOT = {
				tag = ENG
				tag = SPA
				tag = CAN
				tag = RAJ
				tag = AST
				tag = NZL
				tag = NEW
				tag = SAF
				tag = ZIM
				tag = EGY
				tag = JOR
				tag = OMA
			}
		}
		visible = {
			is_in_faction = no
			# is_major = no
			is_puppet = no
			has_war = yes
			NOT = {
				tag = ALY
			}
			OR = {
				tag = BRA
				tag = ARG
				capital_scope = {
					is_on_continent = europe
				}
			}
		}
		available = {
			# threat > 0.2
			NOT = { has_war_with = ALY }
			if = { #If you are a communist country and could join the soviet faction, dont join the allies
				limit = {
					has_government = communism
					SOV = {
						has_far_left_government = yes
						NOT = { has_war_with = ROOT }
						is_faction_leader = yes
						ROOT = {
							any_other_country = {
								is_in_faction = yes
								has_war_with = PREV
								has_war_with = SOV
							}
						}
					}
				}
				custom_trigger_tooltip = {
					tooltip = should_go_with_soviet_faction_tt
					always = no
				}
			}
			else = { #Default for everyone. if you are communist and cant join the soviet faction for whatever reason it will be available.	
				custom_trigger_tooltip = {
					tooltip = has_war_with_a_faction_allies_tt
					any_other_country = {
						has_war_with = ROOT
						is_in_faction = yes
						set_temp_variable = { target_faction = THIS.id }
						ALY = {
							has_war_with = var:target_faction
						}
					}
				}
			}
		}
		ai_will_do = {
			base = 500
			modifier = {
				has_defensive_war = no
				factor = 0
			}
		}
		cost = 0
		complete_effect = {
			hidden_effect = {
				log = "[GetDateText]: [Root.GetName]: Selected decision GEN_join_allies (decision to join the allies)"
				every_country = {
					limit = {
						is_faction_leader = yes
					}
					if = {
						limit = {
							any_of_scopes = {
								array = faction_members
								has_war_with = ROOT
							}
							all_of_scopes = {
								array = faction_members
								NOT = {
									has_war_with = ALY
									ALY = {
										any_allied_country = {
											has_war_with = PREV.PREV
										}
									}
								}
							}
						}
						ROOT = { set_country_flag = will_cause_faction_conflict }
					}
				}
			}
			ALY = {
				country_event = { id = genf.1 }
			}
		}
	}
