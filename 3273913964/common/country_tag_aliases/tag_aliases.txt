# Returns SPR if SPR exists and SPA and SPC dont exist
# Returns SPC if SPC exists and SPA and SPR dont exist
# Returns SPA if SPA exists and SPR and SPC dont exist
# If more than one of the Spain tags exist at the same time then it returns whoever owns Madrid state
# If none of the requirements are met it fallbacks to SPR
ESP = {
	fallback = SPR
	targets = { SPR SPA SPC }

	country_score = {
		base = 1
		modifier = {
			add = 10
			tag = SPR
			country_exists = SPR
			NOT = {
				country_exists = SPA
				country_exists = SPC
			}
		}
		modifier = {
			add = 10
			tag = SPC
			country_exists = SPC

				country_exists = SPR
