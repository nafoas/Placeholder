#SocLibs will always get 1/2 their deserved seats unless they get >12% of the vote, when they'll get 3/4
#MarLibs in 1935 will get a scripted 4 seats in 1935, and otherwise act as the the Len/Fas/FarAut algo
#SocDems also almost got their proportional seats, so they'll just get their scripted 8 seats in 1935 and go with the same algorithm
#LibCons will get 3/2 their deserved seats, unless they get over 5%, when they'll get their deserved seats. If they get over 10%, they'll get 1/2, like the liberals
#DemSocs will get 3/5 their deserved seats unless they get over 40%, where they'll get their deserved seats

	multiply_variable = {
		ENG_faraut_seats = ENG_total_seats
	}
	multiply_variable = {
		ENG_fas_seats = ENG_total_seats
	}
	#rounds
	round_variable = ENG_com_seats
	round_variable = ENG_revsoc_seats
	round_variable = ENG_demsoc_seats
	round_variable = ENG_socdem_seats
	round_variable = ENG_soclib_seats
	round_variable = ENG_marlib_seats
	round_variable = ENG_libcon_seats
	round_variable = ENG_soccon_seats
	round_variable = ENG_aut_seats
	round_variable = ENG_autdes_seats
	round_variable = ENG_faraut_seats
	round_variable = ENG_fas_seats
	#1935 add stuff
	if = {
		limit = {
			date > 1935.1.1
			date < 1936.1.1
		}
		ENG_set_1935_seats = yes
	}
	#Gets new total seat value
	ENG_get_total_seats = yes
	#gets discrepency
	subtract_from_variable = {
		ENG_old_total_seats = ENG_total_seats
	}
	#deals with discrepency, if needed
	if = {
		limit = {
			NOT = {
				check_variable = {
					ENG_old_total_seats = 0
				}
			}
		}
		set_variable = {
			ENG_partial_discrepency = ENG_old_total_seats
		}
		multiply_variable = {
			ENG_partial_discrepency = 0.6 #goes to tories
		}
		round_variable = ENG_partial_discrepency
		if = {
			limit = {
				check_variable = {
					ENG_partial_discrpency > 0 #positive
				}
			}
			subtract_from_variable = {
				ENG_old_total_seats = ENG_partial_discrepency #remainder will be given to labour
			}
			add_to_variable = {
				ENG_soccon_seats = ENG_partial_discrepency
			}
			add_to_variable = {
				ENG_demsoc_seats = ENG_old_total_seats
			}
		}
		#change to subtract, if it does not change behavior, the problem is here
		else = { #negative discrepency, somehow too many seats
			subtract_from_variable = {
				ENG_old_total_seats = ENG_partial_discrepency #remainder will be taken from tories
			}
			add_to_variable = {
				ENG_soccon_seats = ENG_old_total_seats
			}
			add_to_variable = {
				ENG_demsoc_seats = ENG_partial_discrepency
			}
		}
	}
	ENG_get_total_seats = yes
	#gets half
	set_variable = {
		ENG_seats_needed = ENG_total_seats
	}
	divide_variable = {
		ENG_seats_needed = 2
	}
	round_variable = ENG_seats_needed
	#Gets coalition
	set_variable = {
		ENG_coalition_seats = 0
	}
	ENG_get_coalition_seats = yes
}

		ENG_economic_spending_opposite = ENG_economic_intake
	}
	divide_temp_variable = {
		ENG_economic_spending_opposite = 5
	}
	set_temp_variable = {

	}
	multiply_temp_variable = {

	}
	if = {
		limit = {
			check_variable = {
				ENG_economic_spending_opposite > 0 #positive
			}
		}
		clamp_variable = {
			var = ENG_economic_stability
			max = ENG_economic_spending_opposite
			min = ENG_economic_spending_opposite2
		}
	}
	else = {
		clamp_variable = {

			max = ENG_economic_spending_opposite2
			min = ENG_economic_spending_opposite
		}
	}

ENG_micro_depression_recovery = {
	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = ENG_Great_Depression_2
				}
			}
			has_idea = ENG_Great_Depression
		}
		remove_ideas = ENG_Great_Depression
		set_variable = { ENG_Great_Depresion_stab_var = -0.10 }
		set_variable = { ENG_Great_Depresion_indus_factory_var = -0.22 }
		set_variable = { ENG_Great_Depresion_dock_var = -0.20 }
		set_variable = { ENG_Great_Depresion_building_speed_var = -0.22 }
		set_variable = { ENG_Great_Depresion_cosumer_goods_var = 0.067 }
		set_variable = { ENG_Great_Depresion_iterator_var = 1 }
		add_dynamic_modifier = {
			modifier = ENG_Great_Depression_2
		}
	}
	if = {
		limit = {
			check_variable = { ENG_Great_Depresion_iterator_var < 4 }
		}
		add_to_variable = { ENG_Great_Depresion_stab_var = 0.01 }
		add_to_variable = { ENG_Great_Depresion_indus_factory_var = 0.02 }
		add_to_variable = { ENG_Great_Depresion_dock_var = 0.02 }
		add_to_variable = { ENG_Great_Depresion_building_speed_var = 0.02 }
		subtract_from_variable = { ENG_Great_Depresion_cg_iterator = 0.005 }
		add_to_variable = { ENG_Great_Depresion_iterator_var = 1 }
	}
	else_if = {
		limit = {
			has_dynamic_modifier = {
				modifier = ENG_Great_Depression_2
			}
		}
		remove_dynamic_modifier = {
			modifier = ENG_Great_Depression_2
		}
		add_ideas = ENG_Great_Depression3
	}
}
