AST_parl_calc_seats = {
	clear_array = AST_senate_member_array
	clear_array = AST_senate_member_status_array
	clear_array = AST_house_member_array
	clear_array = AST_house_member_status_array
	clear_array = AST_party_status_array_prep_empty
	clear_variable = AST_senate_seat_count
	clear_variable = AST_house_seat_count
	clear_variable = AST_senate_quorum
	clear_variable = AST_house_quorum
	add_to_array = { AST_party_status_array_prep_empty = 0 }
	for_each_loop = {
		array = AST_party_status_array
		set_temp_variable = { bdr_id = v }
		set_temp_variable = { frame_id = i }
		add_to_temp_variable = { frame_id = 1 }
		add_to_array = { AST_party_status_array_prep_empty = v }
		for_loop_effect = {
			start = 0
			end = AST_senate_seat_array^i
			add_to_array = { AST_senate_member_array = frame_id }
			add_to_array = { AST_senate_member_status_array = bdr_id }
			add_to_variable = { AST_senate_seat_count = 1 }
		}
		for_loop_effect = {
			start = 0
			end = AST_house_seat_array^i
			add_to_array = { AST_house_member_array = frame_id }
			add_to_array = { AST_house_member_status_array = bdr_id }
			add_to_variable = { AST_house_seat_count = 1 }
		}
	}
	set_variable = { AST_senate_quorum = AST_senate_seat_count }
	divide_variable = { AST_senate_quorum = 2 }
	round_variable = AST_senate_quorum
	set_variable = { AST_house_quorum = AST_house_seat_count }
	divide_variable = { AST_house_quorum = 2 }
	round_variable = AST_house_quorum
	if = {
		limit = { check_variable = { AST_parl_status = 1 } }
		AST_bill_calc_votes = yes
	}
}

AST_parl_load_bills = {
	set_variable = { AST_bills_visible_array^num = 0 }
	
	add_to_array = { AST_billdb_has_intro_effect = 1 }
	add_to_array = { AST_billdb_has_fail_effect = 1 }
	add_to_array = { AST_bills_allowed_array = 1 }
}

AST_parl_bill_recheck_conditions = {
	clear_array = AST_bills_visible_array
	set_variable = { AST_bills_visible_array^num = 0 }
	for_each_loop = {
		array = AST_bills_allowed_array
		if = {
			limit = {
				meta_trigger = {
					text = {
						AST_bill_[BILL]_visible = yes
					}
					BILL = "[?v]"
				}
			}
			add_to_array = { AST_bills_visible_array = v }
		}
	}
}

AST_bill_calc_votes = {
	set_variable = { AST_bill_senate_votes = 0 }
	set_variable = { AST_bill_house_votes = 0 }
	clr_country_flag = AST_bill_passing
	for_each_loop = {
		array = AST_senate_seat_array
		if = {
			limit = {
				check_variable = { AST_party_status_array^i = 5 }
			}
			add_to_variable = { AST_bill_senate_votes = v }
		}
	}
	for_each_loop = {
		array = AST_house_seat_array
		if = {
			limit = {
				check_variable = { AST_party_status_array^i = 5 }
			}
			add_to_variable = { AST_bill_house_votes = v }
		}
	}
	if = {
		limit = {
			check_variable = {
				var = AST_bill_senate_votes
				value = AST_senate_quorum
				compare = greater_than_or_equals
			}
			check_variable = {
				var = AST_bill_house_votes
				value = AST_house_quorum
				compare = greater_than_or_equals
			}
		}
		set_country_flag = AST_bill_passing
	}
}

AST_parl_daily = {
	if = {
		limit = {
			check_variable = { AST_parl_status = 1 }
		}
		subtract_from_variable = { AST_parl_vote_countdown = 1 }
		if = {
			limit = {
				check_variable = { AST_parl_vote_countdown = 0 }
			}
			AST_bill_calc_votes = yes
			if = {
				limit = {
					has_country_flag = AST_bill_passing
				}
				AST_bill_pass = yes
				else = { AST_bill_fail = yes }
			}
		}
	}
}

AST_bill_pass = {
	remove_from_array = { AST_bills_visible_array = v }
	remove_from_array = { AST_bills_allowed_array = v }
	meta_effect = {
		text = {
			AST_bill_[BILL]_pass = yes
		}
		BILL = "[?AST_parl_selected_bill]"
	}
	set_variable = { AST_parl_status = 2 }
	AST_parl_bill_recheck_conditions = yes
}

AST_bill_fail = {
	if = {
		limit = {
			is_in_array = {
				array = AST.AST_billdb_has_fail_effect
				value = AST_parl_selected_bill
			}
		}
		meta_effect = {
			text = {
				AST_bill_[BILL]_fail = yes
			}
			BILL = "[?AST_parl_selected_bill]"
		}
	}
	set_variable = { AST_parl_status = 2 }
	AST_parl_bill_recheck_conditions = yes
}
