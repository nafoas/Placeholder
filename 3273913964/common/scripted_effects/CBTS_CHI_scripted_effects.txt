CHI_add_communist_influence = {
	
	var:provinceid = {
		add_to_variable = { com_influence = num }
		clamp_variable = {
			var = com_influence
			min = 0
			max = 100
		}

		# Check if all variables sum to 100%
		if = {
			limit = {
				set_temp_variable = { temp_total = com_influence }
				add_to_temp_variable = { temp_total = kmt_influence }
				check_variable = { temp_total > 100 }
			}
			set_temp_variable = { to_reduce = temp_total }
			subtract_from_temp_variable = { to_reduce = 100 }
			subtract_from_variable = { kmt_influence = to_reduce }
		}
	}

}
