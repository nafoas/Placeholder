# Reduce Effects of the Great Depression
HUN_effects_of_the_great_depression_level_down = {
	if = {
		limit = {
			has_idea = HUN_effects_of_the_great_depression_5
		}
		swap_ideas = {
			remove_idea = HUN_effects_of_the_great_depression_5
			add_idea = HUN_industrializing_society
		}
	}
	else_if = {
		limit = {
			has_idea = HUN_effects_of_the_great_depression_1
		}
		swap_ideas = {
			remove_idea = HUN_effects_of_the_great_depression_1
			add_idea = HUN_effects_of_the_great_depression_2
		}
	}
	else_if = {
		limit = {
			has_idea = HUN_effects_of_the_great_depression_2
		}
		swap_ideas = {
			remove_idea = HUN_effects_of_the_great_depression_2
			add_idea = HUN_effects_of_the_great_depression_3
		}
	}
	else_if = {
		limit = {
			has_idea = HUN_effects_of_the_great_depression_3
		}
		swap_ideas = {
			remove_idea = HUN_effects_of_the_great_depression_3
			add_idea = HUN_effects_of_the_great_depression_4
		}
	}
	else_if = {
		limit = {
			has_idea = HUN_effects_of_the_great_depression_4
		}
		swap_ideas = {
			remove_idea = HUN_effects_of_the_great_depression_4
			add_idea = HUN_effects_of_the_great_depression_5
		}
	}
}

#Debt Check Effect 
HUN_debt_check_effect = {
	hidden_effect = {
		if = {
			limit = {
				check_variable = { var = HUN_debt_to_the_lon_var value = 0 compare = equals }
			}
			set_variable = { HUN_debt_to_the_lon_paid_var = 10 }
		}
		if = {
			limit = {
				check_variable = { var = HUN_debt_to_the_lon_paid_var value = 10 compare = greater_than_or_equals }
			}
			remove_ideas = HUN_debt_to_the_lon_3
			clear_variable = HUN_debt_to_the_lon_paid_var
			clear_variable = HUN_debt_to_the_lon_var
			set_country_flag = HUN_completely_debt_free
		}
		else_if = {
			limit = {
				check_variable = { var = HUN_debt_to_the_lon_paid_var value = 6 compare = greater_than_or_equals }
				check_variable = { var = HUN_debt_to_the_lon_paid_var value = 10 compare = less_than }
			}
			swap_ideas = {
				remove_idea = HUN_debt_to_the_lon_2
				add_idea = HUN_debt_to_the_lon_3
			}
		}
		else_if = {
			limit = {
				check_variable = { var = HUN_debt_to_the_lon_paid_var value = 4 compare = greater_than_or_equals }
				check_variable = { var = HUN_debt_to_the_lon_paid_var value = 10 compare = less_than }
			}
			swap_ideas = {
				remove_idea = HUN_debt_to_the_lon
				add_idea = HUN_debt_to_the_lon_2
			}
		}
	}
}
