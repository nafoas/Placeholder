# The following scripted effect does the main job of generating the gui, by adding the seats for each party in a grid.
# This should be called every time you need to update the gui, when you add/subtract seats for example.
# Read the comments if you're interested in how it works
gen_reload_parliament_gui = {
	# Clears what's currently on the gui, if there's anything there.
	clear_array = gen_parliament_main_array

	# The array gen_temp_seat is duplicated from the array gen_parliament_seat_array.
	for_each_loop = {
		array = gen_parliament_seat_array

		add_to_array = { gen_temp_seat = gen_parliament_seat_array^i }
	}

	# It's a suprise tool for later
	set_temp_variable = { gen_temp_index_1 = 0 }
	set_temp_variable = { gen_temp_index_2 = 1 }

	# Set gen_temp_size as the highest index in the array, highest index is the highest value x in gen_temp_array^x.
	set_temp_variable = { gen_temp_size = gen_temp_seat^num }
	subtract_from_temp_variable = { gen_temp_size = 1 }

	# What this does is loop through the gen_temp_seat array, adding the current value to the next value on the list.
	# gen_temp_seat^0 is the first value on the list, gen_temp_seat^1 is the next and so forth.
	# the loop ends when gen_temp_index_1 is equal to the size of the array - 1 (basically the max index).
	while_loop_effect = {
		limit = {
			check_variable = { gen_temp_index_1 < gen_temp_size }
		}
		# The suprise tool starts out as gen_temp_seat^0 and gen_temp_seat^1 and then is incremented every loop , adding the current value to the next value on the list.
		add_to_variable = { gen_temp_seat^gen_temp_index_2 = gen_temp_seat^gen_temp_index_1 }
		add_to_temp_variable = { gen_temp_index_1 = 1 }
		add_to_temp_variable = { gen_temp_index_2 = 1 }
	}

	# resize_array fills the gen_parliament_member_array with the id of the parliament member x amount of times where x is the number of seats the member has.
	# so the array will look like this basically:
	# 1,1,1,1,1,1,1,...(repeated 90 times),2,2,2,2,2,2,2(repeated 35 times),3,3,3.. etc.
	for_loop_effect = {
		start = 0
		end = gen_temp_seat^num

		resize_array = { array = gen_parliament_main_array value = gen_parliament_member_array^v size = gen_temp_seat^v }
	}

	# Clear the gen_temp_seat array used previously since it's not needed anymore.
	clear_array = gen_temp_seat
}

# Bonus code for adding/subtracting seats
gen_modify_parliament_seat = {
	subtract_from_temp_variable = { gen_temp_member = 1 }

	# check to see if result overflows
	set_temp_variable = { gen_temp = gen_parliament_seat_array^gen_temp_member }
	add_to_temp_variable = { gen_temp = gen_temp_seats }

	if = {
		limit = {
			check_variable = { gen_temp > gen_parliament_main_array^num }
		}
		subtract_from_temp_variable = { gen_temp = gen_parliament_main_array^num }
		subtract_from_temp_variable = { gen_temp_seats = gen_temp }
	}
	else_if = {
		limit = {
			check_variable = { gen_temp < 0 }
		}
		multiply_temp_variable = { gen_temp = -1 }
		subtract_from_temp_variable = { gen_temp_seats = gen_temp }
	}

	# adds all parties to gen_temp_array
	for_loop_effect = {
		start = 0
		end = gen_parliament_seat_array^num

		add_to_temp_array = { gen_temp_array = v }
	}

	# remove from list of parties the party to be added/subtracted
	remove_from_temp_array = { array = gen_temp_array value = gen_temp_member }

	# give party new seat value
	add_to_variable = { gen_parliament_seat_array^gen_temp_member = gen_temp_seats }

	# if its add
	if = {
		limit = {
			check_variable = { gen_temp_seats > 0 }
		}
		# loop until seats to be added is 0
		while_loop_effect = {
			limit = {
				check_variable = { gen_temp_seats > 0 }
			}

			# loops through remaining parties, subtracting 1 seat each time
			for_loop_effect = {
				start = 0
				end = gen_temp_array^num

				if = {
					limit = {
						check_variable = { gen_parliament_seat_array^gen_temp_array^v > 0 } # party must have more than 0 seats to be subtracted
						check_variable = { gen_temp_seats > 0 } # seats to be added is bigger than 0 currently
					}
					subtract_from_variable = { gen_parliament_seat_array^gen_temp_array^v = 1 } # subtract from party 1 seat
					subtract_from_temp_variable = { gen_temp_seats = 1 } # subtract from seats to be added
				}
			}
		}
	}
	# if its subtract
	else_if = {
		limit = {
			check_variable = { gen_temp_seats < 0 }
		}
		# loop until seats to be subtracted is 0
		while_loop_effect = {
			limit = {
				check_variable = { gen_temp_seats < 0 }
			}

			# loops through remaining parties, adding 1 seat each time
			for_loop_effect = {
				start = 0
				end = gen_temp_array^num

				if = {
					limit = {
						check_variable = { gen_parliament_seat_array^gen_temp_array^v < gen_parliament_main_array^num } # party must have less than max seats to be added
						check_variable = { gen_temp_seats < 0 } # seats to be subtracted is less than 0 currently
					}
					add_to_variable = { gen_parliament_seat_array^gen_temp_array^v = 1 } # add to party 1 seat
					add_to_temp_variable = { gen_temp_seats = 1 } # add to seats to be subtracted
				}
			}
		}
	}
	# do nothing if 0

	# Modify what shows up in the gui
	gen_reload_parliament_gui = yes
}
