WGR_economic_stability_check = { #(-2*spending - intake)/5
	set_temp_variable = {
		WGR_economic_spending_opposite = WGR_economic_spending
	}
	multiply_temp_variable = {
		WGR_economic_spending_opposite = -2
	}
	subtract_from_temp_variable = {
		WGR_economic_spending_opposite = WGR_economic_intake
	}
	divide_temp_variable = {
		WGR_economic_spending_opposite = 5
	}
	set_temp_variable = {
		WGR_economic_spending_opposite2 = WGR_economic_spending_opposite
	}
	multiply_temp_variable = {
		WGR_economic_spending_opposite2 = -1
	}
	if = {
		limit = {
			check_variable = {
				WGR_economic_spending_opposite > 0 #positive
			}
		}
		clamp_variable = {
			var = WGR_economic_stability
			max = WGR_economic_spending_opposite
			min = WGR_economic_spending_opposite2
		}
	}
	else = {
		clamp_variable = {
			var = WGR_economic_stability
			max = WGR_economic_spending_opposite2
			min = WGR_economic_spending_opposite
		}
	}
}
