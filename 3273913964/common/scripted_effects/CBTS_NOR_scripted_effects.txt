NOR_add_governing_seats = {
	set_country_flag = NOR_voting
	custom_effect_tooltip = will_vote
	if = {
		limit = { has_government = communism }
		set_variable = { NOR_supseat = NOR_com_seats }
	}
	if = {
		limit = { 
			OR = {
				has_government = social_democracy
				has_government = socialism
				has_government = authoritarian_socialism
			}
		}
		set_variable = { NOR_supseat = NOR_socdem_seats }
	}
	if = {
		limit = {
			OR = {
				has_government = market_liberalism
				has_government = social_liberalism
			}
		}
		set_variable = { NOR_supseat = NOR_soclib_seats }
	}
	if = {
		limit = {
			OR = {
				has_government = jacobin
				has_government = neutrality
			}
		}
		set_variable = { NOR_supseat = NOR_libcon_seats }
	}
	if = {
		limit = { has_government = democratic }
		set_variable = { NOR_supseat = NOR_soccon_seats }
	}
	if = {
		limit = { has_government = monarchism }
		set_variable = { NOR_supseat = NOR_mon_seats }
	}
	if = {
		limit = { has_government = nationalism }
		set_variable = { NOR_supseat = NOR_nat_seats }
	}
	if = {
		limit = { has_government = fascism }
		set_variable = { NOR_supseat = NOR_fas_seats }
	}
}

	set_temp_variable = { NOR_total_popularity = 0 }

	## Here it takes the party's popularity
	if = {

		}

		multiply_temp_variable = { communist_popularity = 100 }
		add_to_temp_variable = { NOR_total_popularity = communist_popularity }
	}
	if = {
		limit = {
			NOT = { has_country_flag = socdem_ban }
		}
