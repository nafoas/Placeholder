RUS_Get_Soc_Pop = {
	set_variable = {
		RUS_Soc_Pop = party_popularity@socialism
	}
	add_to_variable = {
		RUS_Soc_Pop = party_popularity@authoritarian_socialism
	}
}
RUS_Get_Dem_Pop = {
	set_variable = {
		RUS_Dem_Pop = party_popularity@social_liberalism
	}
	if = {
		limit = {
			has_country_flag = RUS_Trud_Return
		}
		add_to_variable = {
			RUS_Dem_Pop = party_popularity@social_democracy
		}
	}
	if = {
		limit = {
			has_country_flag = RUS_PSR_Return
			NOT = {
				has_variable = RUS_socialist_influence
			}
		}
		add_to_variable = {
			RUS_Dem_Pop = party_popularity@socialism
		}
	}
	if = {
		limit = {
			has_country_flag = RUS_Prog_Return
		}
		add_to_variable = {
			RUS_Dem_Pop = party_popularity@market_liberalism
		}
	}
	if = {
		limit = {
			has_country_flag = RUS_Octo_Return
		}
		add_to_variable = {
			RUS_Dem_Pop = party_popularity@jacobin
		}
		add_to_variable = {
			RUS_Dem_Pop = party_popularity@democratic
		}
	}
}
RUS_add_influence_socialists = {
	if = {
		limit = {
			check_variable = {
				RUS_socialist_influence < 100
			}
		}
		add_to_variable = {
			RUS_socialist_influence = RUS_add_influence_value #adds the passed value to the influence
		}
		clamp_variable = {
			var = RUS_socialist_influence # clamps it between 0 and 100
			min = 0
			max = 100
		}
		set_temp_variable = {
			faction_number = 0 #this next section counts the amount of factions you have
		}
		if = {
			limit = {
				has_variable = RUS_democrat_influence
				check_variable = {
					RUS_democrat_influence > 0
				}
			}
			add_to_temp_variable = {
				faction_number = 1
			}
		}
		if = {
			limit = {
				has_variable = RUS_vlasov_influence
				check_variable = {
					RUS_vlasov_influence > 0
				}
			}
			add_to_temp_variable = {
				faction_number = 1
			}
		}
		if = {
			limit = {
				has_variable = RUS_monarchist_influence
				check_variable = {
					RUS_monarchist_influence > 0
				}
			}
			add_to_temp_variable = {
				faction_number = 1
			}
		}
		if = {
			limit = {
				has_variable = RUS_nationalist_influence
				check_variable = {
					RUS_nationalist_influence > 0
				}
			}
			add_to_temp_variable = {
				faction_number = 1
			}
		}
		if = {
			limit = { #if there is no opposition, nothing happens
				check_variable = {
					faction_number = 0
				}
			}
		}
		else = { #otherwise, it divides by the faction number, rounds to two decimal places, and subtracts it from the other influences
			divide_variable = {
				RUS_add_influence_value = faction_number
			}
			multiply_variable = {
				RUS_add_influence_value = 100
			}
			round_variable = RUS_add_influence_value
			divide_variable = {
				RUS_add_influence_value = 100
			}
			if = {
				limit = {
					has_variable = RUS_democrat_influence
					check_variable = {
						RUS_democrat_influence > 0
					}
				}
				subtract_from_variable = {
					RUS_democrat_influence = RUS_add_influence_value
				}
				clamp_variable = {
					var = RUS_democrat_influence # clamps it between 0 and 100
					min = 0
					max = 100
				}
			}
			if = {
				limit = {
					has_variable = RUS_vlasov_influence
					check_variable = {
						RUS_vlasov_influence > 0
					}
				}
				subtract_from_variable = {
					RUS_vlasov_influence = RUS_add_influence_value
				}
				clamp_variable = {
					var = RUS_vlasov_influence # clamps it between 0 and 100
					min = 0
					max = 100
				}
			}
			if = {
				limit = {
					has_variable = RUS_monarchist_influence
					check_variable = {
						RUS_monarchist_influence > 0
					}
				}
				subtract_from_variable = {
					RUS_monarchist_influence = RUS_add_influence_value
				}
				clamp_variable = {
					var = RUS_monarchist_influence # clamps it between 0 and 100
					min = 0
					max = 100
				}
			}
			if = {
				limit = {
					has_variable = RUS_nationalist_influence
					check_variable = {
						RUS_nationalist_influence > 0
					}
				}
				subtract_from_variable = {
					RUS_nationalist_influence = RUS_add_influence_value
				}
				clamp_variable = {
					var = RUS_nationalist_influence # clamps it between 0 and 100
					min = 0
					max = 100
				}
			}
		}
		set_variable = {
			RUS_add_influence_value = 0
		}
		RUS_check_for_hundred = yes
		RUS_Get_Soc_Pop = yes
		RUS_Get_Dem_Pop = yes
	}
}
