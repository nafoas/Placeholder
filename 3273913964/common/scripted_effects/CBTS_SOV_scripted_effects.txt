SOV_Congress_Finish = { #selects winner
	custom_effect_tooltip = SOV_congress_over
	if = {
		limit = {
			check_variable = {
				SOV_Congress_Points > 3
			}
		}
		hidden_effect = {
			country_event = {
				id = sov_civil_war.22 #Bukharin
				hours = 1
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				SOV_Congress_Points < -3
			}
		}
		hidden_effect = {
			country_event = {
				id = sov_civil_war.21 #Zinoviev
				hours = 1
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				SOV_Congress_Points = 0
			}
			NOT = {
				has_country_flag = SOV_Red_Army_Defects
			}
			has_country_flag = SOV_Zin_Available
			NOT = {
				has_country_flag = SOV_stop_bukharinists
			}
		}
		hidden_effect = {
			country_event = {
				id = sov_civil_war.23 #Army
				days = 1
			}
		}	
	}
	else = {
		hidden_effect = {
			country_event = {
				id = sov_civil_war.20 #malenkov
				hours = 1
			}
		}
	}
}

SOV_figure_out_2rcw = {
	if = {
		limit = {
			OR = {
				is_historical_focus_on = yes
				has_game_rule = {
					rule = SOV_2rcw_enabled
					option = SOV_2RCW_DISABLED
				}
			}
		}
		log = "Disabled 2rcw: Either hist focuses are on or disabled through game rules" #No 2rcw
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = SOV_2rcw_enabled
				option = SOV_2RCW_DEFAULT
			}
		}
		random_list = {
			15 = {
				set_country_flag = SOV_enable_2rcw
				log = "15% chance of 2rcw: YES"
			}
			85 = {
				log = "15% chance of 2rcw: NO"
			}
		}
	}
	else = {
		set_country_flag = SOV_enable_2rcw
		log = "Enabled 2RCW because of game rule"
	}
}
