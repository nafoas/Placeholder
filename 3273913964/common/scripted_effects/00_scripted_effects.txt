increase_spending = {
	if = {
		limit = {
			has_idea = very_high_research_spending
		}
		add_command_power = 10
	}
	if = {
		limit = {
			has_idea = high_research_spending
		}
		swap_ideas = {
			remove_idea = high_research_spending
			add_idea = very_high_research_spending
		}
	}
	if = {
		limit = {
			has_idea = moderate_research_spending
		}
		swap_ideas = {
			remove_idea = moderate_research_spending
			add_idea = high_research_spending
		}
	}
	if = {
		limit = {
			has_idea = minimal_research_spending
		}
		swap_ideas = {
			remove_idea = minimal_research_spending
			add_idea = low_research_spending
		}
	}
	if = {
		limit = {
			has_idea = no_research_spending
		}
		swap_ideas = {
			remove_idea = no_research_spending
			add_idea = minimal_research_spending
		}
	}
}

#expectes a temp variable country_to_initiate which should contain original_tag to instantiate as a collabration government
instantiate_collaboration_government = {
	custom_effect_tooltip = compliance_80_effect_tooltip
	hidden_effect = {
		set_variable = { collaboration_formed@var:country_to_initiate = 1 }
		
		if = {
			# if country_to_initiate does not exist, simply release it as a puppet
			limit = { 
				var:country_to_initiate = {
					exists = no
				}
			}
			var:country_to_initiate = { set_variable = { collaboration_formed_by = PREV } }
			release_autonomy = {
				target = country_to_initiate
				autonomy_state = autonomy_collaboration_government
				freedom_level = 0.5
				release_non_owned_controlled = yes
			}
		}
		else = {
			# if country_to_initiate exists, create a dynamic country as our new puppet
			
			create_dynamic_country = {
				original_tag = country_to_initiate
			
				set_variable = { collaboration_formed_by = PREV }
				set_temp_variable = { new_country = this }
				
				PREV = {
					every_controlled_state = {
						limit = { occupied_country_tag = country_to_initiate }
						
						var:new_country = {
							transfer_state = PREV
						}
					}
					
					puppet = new_country
					
					set_autonomy = {
						target = new_country
						autonomy_state = autonomy_collaboration_government
						freedom_level = 0.5
					}
				}
			}
		}
	}
}
