#Chooses which party wins the election
# 0 - Left Victory (Com, DemSoc, SocDem)
# 1 - Center Left (SocDem, SocLib, MarLib)
# 2 - Center Right (MarLib, LibCon, SocCon)
# 3 - Right (SocCon, Aut, Mon, Nat)
GER_Goerdeler_Set_Victorious_Party = {
	set_variable = {
		Winning_Group = random
	}
	multiply_variable = {
		Winning_Group = 3
	}
	if = {
		limit = {
			has_stability > 0.9
		}
		add_to_variable = {
			Winning_Group = 0.5
		}
	}
	else_if = {
		limit = {
			has_stability > 0.6
		}
		add_to_variable = {
			Winning_Group = 0.15
		}
	}
	else_if = {
		limit = {
			has_stability < 0.1
		}
		add_to_variable = {
			Winning_Group = -0.5
		}
	}
	else_if = {
		limit = {
			has_stability < 0.4
		}
		add_to_variable = {
			Winning_Group = -0.15
		}
	}
	if = {
		limit = {
			has_war_support > 0.8
		}
		add_to_variable = {
			Winning_Group = 0.6
		}
	}
	else_if = {
		limit = {
			has_war_support > 0.6
		}
		add_to_variable = {
			Winning_Group = 0.2
		}
	}
	else_if = {
		limit = {
			has_war_support < 0.3
		}
		add_to_variable = {
			Winning_Group = -0.2
		}
	}
	if = { #inflation moves towards center
		limit = {
			has_idea = GER_DNVP_Inflation
		}
		if = {
			limit = {
				check_variable = {
					Winning_Group > 2.49
				}
			}
			add_to_variable = {
				Winning_Group = -0.5
			}
		}
		if = {
			limit = {
				check_variable = {
					Winning_Group < 1.5
				}
			}
			add_to_variable = {
				Winning_Group = 0.5
			}
		}
	}
}
