    if = {
	    limit = {
		    has_idea = great_depression4
		}
		remove_ideas = great_depression4
	}
	if = {
	    limit = {
		    has_idea = great_depression3
		}
		swap_ideas = { remove_idea = great_depression3 add_idea = great_depression4 }
	}
	if = {
	    limit = {
		    has_idea = great_depression2
		}
		swap_ideas = { remove_idea = great_depression2 add_idea = great_depression3 }
	}

		}
		swap_ideas = { remove_idea = great_depression add_idea = great_depression2 }
	}
}
