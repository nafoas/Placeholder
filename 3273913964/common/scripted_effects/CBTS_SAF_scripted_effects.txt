				has_government = fascism
			}
			add_to_variable = {
				SAF_fas_seats = SAF_old_total_seats
			}
		}

		SAF_seats_needed = SAF_total_seats

		SAF_seats_needed = 2

		SAF_coalition_seats = 0
	}
