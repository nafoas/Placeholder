COALITION_set_up_array = {
	if = {
		limit = {
			has_government = communism
		}
		add_to_array = {
			array = coalition_party_list
			value = 0
		}
	}
	else_if = {
		limit = {
			has_government = authoritarian_socialism
		}
		add_to_array = {
			array = coalition_party_list
			value = 1
		}
	}
	else_if = {
		limit = {
			has_government = socialism
		}
		add_to_array = {
			array = coalition_party_list
			value = 2
		}
	}
	else_if = {
		limit = {
			has_government = social_democracy
		}
		add_to_array = {
			array = coalition_party_list
			value = 3
		}
	}
	else_if = {
		limit = {
			has_government = social_liberalism
		}
		add_to_array = {
			array = coalition_party_list
			value = 4
		}
	}
	else_if = {
		limit = {
			has_government = market_liberalism
		}
		add_to_array = {
			array = coalition_party_list
			value = 5
		}
	}
	else_if = {
		limit = {
			has_government = jacobin
		}
		add_to_array = {
			array = coalition_party_list
			value = 6
		}
	}
	else_if = {
		limit = {
			has_government = democratic
		}
		add_to_array = {
			array = coalition_party_list
			value = 7
		}
	}
	else_if = {
		limit = {
			has_government = monarchism
		}
		add_to_array = {
			array = coalition_party_list
			value = 8
		}
	}
	else_if = {
		limit = {
			has_government = neutrality
		}
		add_to_array = {
			array = coalition_party_list
			value = 9
		}
	}
	else_if = {
		limit = {
			has_government = nationalism
		}
		add_to_array = {
			array = coalition_party_list
			value = 10
		}
	}
	else_if = {
		limit = {
			has_government = fascism
		}
		add_to_array = {
			array = coalition_party_list
			value = 11
		}
	}
	COALITION_POP_NUMBER = yes
}

#SET COALITION SEATS. For Parliaments
set_coalition_seats = {
	set_variable = { colseat = 0 }
	
	if = {
		limit = {
			is_in_array = {
				array = coalition_party_list
				value = 0
		   }
		}
		add_to_variable = { var = colseat value = communist_seats }
	}
	
	if = {
		limit = {
			OR = {
				is_in_array = {
					array = coalition_party_list
					value = 1
				}
				is_in_array = {
					array = coalition_party_list
