GCW_is_ongoing = {
	has_global_flag = GCW_ongoing_flag
	OR = {
		country_exists = GER
		country_exists = GRM
		country_exists = DDR
		country_exists = WGR
	}
}

GER_exists = {
	country_exists = GER
	NOT = {
		has_global_flag = GCW_ongoing_flag
	}
}
DDR_exists = {
	country_exists = DDR
	NOT = {
		has_global_flag = GCW_ongoing_flag
	}
}
WGR_exists = {
	country_exists = WGR
	NOT = {
		has_global_flag = GCW_ongoing_flag
	}
}

only_one_germany_exists = {
	OR = {
		AND = {
			country_exists = GER
			NOT = {
				country_exists = WGR
				country_exists = DDR
			}
		}
		AND = {
			country_exists = WGR
			NOT = {
				country_exists = GER
				country_exists = DDR
			}
		}
		AND = {
			country_exists = DDR
			NOT = {
				country_exists = WGR
				country_exists = GER
			}
		}
	}
}

		has_wargoal_against = SOV
	}
}
