agrarians_highest_influence = {
    OR = {
        check_variable = { CZE_agrarian_influence > CZE_christian_democrat_influence }
        AND = {
            check_variable = { CZE_agrarian_influence = CZE_christian_democrat_influence }
            check_variable = { CZE.party_popularity@democratic = CZE.party_popularity@jacobin }
        }
    }
    OR = {
        check_variable = { CZE_agrarian_influence > CZE_national_democrat_influence }
        AND = {
            check_variable = { CZE_agrarian_influence = CZE_national_democrat_influence }
            if = {
                limit = { has_country_flag = national_democrats_are_rightauth }
                check_variable = { CZE.party_popularity@democratic > CZE.party_popularity@nationalism }
            }
            else = {
                always = no
            }
        }
    }
    OR = {
        check_variable = { CZE_agrarian_influence > CZE_national_socialist_influence }
        AND = {
            check_variable = { CZE_agrarian_influence = CZE_national_socialist_influence }
            check_variable = { CZE.party_popularity@democratic > CZE.party_popularity@social_liberalism }
        }
    }
    OR = {
        check_variable = { CZE_agrarian_influence > CZE_social_democrat_influence }
        AND = {
            check_variable = { CZE_agrarian_influence = CZE_social_democrat_influence }
            check_variable = { CZE.party_popularity@democratic > CZE.party_popularity@social_democracy }
        }
    }
    OR = {
        check_variable = { CZE_agrarian_influence > CZE_traders_influence }
        AND = {
            check_variable = { CZE_agrarian_influence = CZE_traders_influence }
            check_variable = { CZE.party_popularity@democratic > CZE.party_popularity@market_liberalism }
        }
    }
}
