# ============================================================================
# Vanilla 1.19 scripted-trigger compatibility shims (BUGFIX PASS, cross-cutting)
# ============================================================================
# WHY THIS FILE EXISTS
# The mod's descriptor.mod sets  replace_path="common/scripted_triggers" , which
# REPLACES the entire vanilla (and underlying base-mod) scripted_triggers folder
# with the mod's own set. As a result, a number of scripted triggers that are
# DEFINED by the base game / base content but CALLED by base files the mod does
# NOT override (and therefore still load at runtime) became "Invalid trigger".
#
# This file restores ONLY the base-vanilla 1.19 scripted triggers that are:
#   (a) confirmed to be byte-identical across every pristine-vanilla and
#       confirmed-1.19 mirror that defines them, AND
#   (b) free of any dependency on map state-IDs, country tags, or ideas that the
#       mod may have reworked (so restoring them cannot change the mod's design
#       or balance).
# The remaining undefined triggers require a source/scope decision and are
# itemised in PORTING/BUGFIX-PASS/ESCALATIONS.md (they are NOT guessed here).
#
# SOURCES (pristine / confirmed-1.19 vanilla, cross-referenced):
#  - should_play_south_american_music :
#       vanilla common/scripted_triggers/00_scripted_triggers.txt
#       Verified identical in: Road-to-56-RP/roadto56rp, deliciousmods/1956_beta,
#       and ~25 other confirmed-current mirrors (all: capital_scope on south_america).
#  - can_unlock_second_track_of_sf_doctrine :
#       vanilla common/scripted_triggers/_special_forces_scripted_triggers.txt
#       Verified identical in: Kaiserreich/Kaiserreich-HOI4 (confirmed 1.19), plus a
#       repo literally named "(vanilla) 00_scripted_triggers.txt" (Kama-Pushka/Kursach-Fizika).
# ============================================================================

should_play_south_american_music = {
	capital_scope = { is_on_continent = south_america }
}

can_unlock_second_track_of_sf_doctrine = {
	if = {
		limit = { check_variable = { sf_trees > 1 } }
		custom_trigger_tooltip = {
			tooltip = can_unlock_second_track_special_forces_doctrine_yes
			check_variable = { sf_trees > 1 }
		}
	}
	else = {
		custom_trigger_tooltip = {
			tooltip = can_unlock_second_track_special_forces_doctrine_no
			always = no
		}
	}
}
