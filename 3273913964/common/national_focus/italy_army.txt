# Spawns 2 units
shared_focus = {
	id = ITA_celere_division
	icon = GFX_focus_Light_Tanks
	prerequisite = {
		focus = ITA_increase_army_spending
	}
	prerequisite = {
		focus = ITA_entrench_the_party
	}
	x = 0
	y = 4
	relative_position_id = ITA_reorganize_the_command
	cost = 4
	ai_will_do = {
		factor = 1000
	}
	available = {
		controls_state = 2
	}
	completion_reward = {
		custom_effect_tooltip = ITA_celere_division_tt
		hidden_effect = {
			division_template = {
				name = "Divisione Celere" 		# Divisione Celere (Fast Division)
				regiments = {
					cavalry = { x = 0 y = 0 }
					cavalry = { x = 0 y = 1 }
					cavalry = { x = 1 y = 0 }
					cavalry = { x = 1 y = 1 }
					motorized = { x = 2 y = 0 }		# (elite) Bersaglieri Regiment
					motorized = { x = 2 y = 1 }
					light_armor = { x = 3 y = 0 }	# L3/35 Tankette Battaltion
				}
				priority = 2
			}
			## 6a Armata ##
			# Corpo d'Armata Celere #
			if = {
				limit = {
					controls_state = 160
				}
				160 = {
					create_unit = {
						division = "name = \"1. Divisione Celere 'Eugenio di Savoia'\" division_template = \"Divisione Celere\" start_experience_factor = 0.1 start_equipment_factor = 1.0"
						owner = ITA
						prioritize_location = 11584
					}
				}
			}
			else = {
				random_controlled_state = {
					limit = {
						is_in_home_area = yes
					}
					create_unit = {
						division = "name = \"1. Divisione Celere 'Emanuele Filiberto'\" division_template = \"Divisione Celere\" start_experience_factor = 0.1 start_equipment_factor = 1.0"
						owner = ITA
					}
				}
			}

			if = {
				limit = {
					controls_state = 160
				}
				160 = {
					create_unit = {
						division = "name = \"2. Divisione Celere\" division_template = \"Divisione Celere\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
						owner = ITA
						prioritize_location = 11584
					}
				}
			}
			else = {
				random_controlled_state = {
					limit = {
						is_in_home_area = yes
					}
					create_unit = {
						division = "name = \"2. Divisione Celere\" division_template = \"Divisione Celere\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
						owner = ITA
					}
				}
			}
		}
	}
}
