# Since index starts as 0, i can get the current seat number by jut adding it by one and then displaying it.
defined_text = {
	name = GetParliamentSeatName
	text = {
		trigger = {
			set_temp_variable = { gen_temp_seat = parliament_main_index }
			add_to_temp_variable = { gen_temp_seat = 1 }
			check_variable = { gen_temp_seat > 0 }
		}
		localization_key = gen_parliament_seat_text
	}
}
