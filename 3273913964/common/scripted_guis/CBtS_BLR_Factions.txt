scripted_gui = {
	BLR_Faction_Button = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "BLR_faction_button_container"
		visible = {
            tag = BLR
			#has_global_flag = Second_Russian_Civil_War
		}
		effects = {
			faction_button_click = {
				if = {
					limit = {
						has_country_flag = BLR_show_factions
					}
					clr_country_flag = BLR_show_factions
				}
				else = {
					set_country_flag = BLR_show_factions
					BLR_Get_Aut_Pop = yes
					BLR_Get_Nat_Pop = yes
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = BLR_show_blr_nationalist
							has_country_flag = BLR_show_autonomist
							has_country_flag = BLR_show_triunionist
						}
					}
					set_country_flag = BLR_show_bulak
				}
			}
		}
		ai_enabled = {
			tag = BLR
			has_global_flag = Second_Russian_Civil_War
		}
		ai_test_interval = 240
		ai_test_variance = 0
		ai_check = {
			tag = BLR
			has_political_power > 39.99
			OR = {
				NOT = {
					has_idea = RUS_faction_influence_block
					has_idea = RUS_rally_block
				}
			}
		}
		ai_weights = {
			faction_button_click = {
				ai_will_do = {
					base = -1
					modifier = {
						NOT = {
							has_idea = RUS_faction_influence_block
							has_idea = RUS_rally_block
						}
						NOT = {
							has_country_flag = BLR_AI_Faction_Open
						}
						add = 100
					}
				}
			}
		}
	}
