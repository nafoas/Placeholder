scripted_gui = {
	RFP_Faction_Button = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "RFP_faction_button_container"
		visible = {
            tag = RFP
			has_global_flag = Second_Russian_Civil_War
		}
		effects = {
			faction_button_click = {
				if = {
					limit = {
						has_country_flag = RFP_show_factions
					}
					clr_country_flag = RFP_show_factions
				}
				else = {
					set_country_flag = RFP_show_factions
					RFP_Get_Rod_Pop = yes
					RFP_Get_Von_Pop = yes
					RFP_Get_Mon_Pop = yes
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = RFP_show_rod
							has_country_flag = RFP_show_von
							has_country_flag = RFP_show_cos
						}
					}
					set_country_flag = RFP_show_rod
				}
			}
		}
		ai_enabled = {
			tag = RFP
			has_global_flag = Second_Russian_Civil_War
		}
		ai_test_interval = 240
		ai_test_variance = 0
		ai_check = {
			tag = RFP
			has_political_power > 39.99
			OR = {
				NOT = {
					has_idea = RUS_faction_influence_block
					has_idea = RUS_rally_block
				}
			}
		}
		ai_weights = {
			faction_button_click = {
				ai_will_do = {
					base = -1
					modifier = {
						NOT = {
							has_idea = RUS_faction_influence_block
							has_idea = RUS_rally_block
						}
						NOT = {
							has_country_flag = RFP_AI_Faction_Open
						}
						add = 100
					}
				}
			}
		}
	}
}
