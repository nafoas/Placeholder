scripted_gui = {
	RUS_Faction_Button = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "faction_button_container"
		visible = {
            tag = RUS
			has_global_flag = Second_Russian_Civil_War
		}
		effects = {
			faction_button_click = {
				if = {
					limit = {
						has_country_flag = RUS_show_factions
					}
					clr_country_flag = RUS_show_factions
				}
				else = {
					set_country_flag = RUS_show_factions
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = RUS_show_socs
							has_country_flag = RUS_show_dems
							has_country_flag = RUS_show_konr
							has_country_flag = RUS_show_mons
							has_country_flag = RUS_show_nats
						}
					}
					set_country_flag = RUS_show_konr
				}
			}
		}
		ai_enabled = {
			tag = RUS
			has_global_flag = Second_Russian_Civil_War
		}
		ai_test_interval = 240
		ai_test_variance = 0
		ai_check = {
			tag = RUS
			has_political_power > 39.99
			OR = {
				NOT = {
					has_idea = RUS_faction_influence_block
					has_idea = RUS_rally_block
				}
			}
		}
		ai_weights = {
			faction_button_click = {
				ai_will_do = {
					base = -1
					modifier = {
						NOT = {
							has_idea = RUS_faction_influence_block
							has_idea = RUS_rally_block
						}
						NOT = {
							has_country_flag = RUS_AI_Faction_Open
						}
						add = 100
					}
				}
			}
		}
	}
