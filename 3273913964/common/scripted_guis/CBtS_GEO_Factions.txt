scripted_gui = {
	GEO_Faction_Button = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "GEO_faction_button_container"
		visible = {
            tag = GEO
			#has_global_flag = Second_Russian_Civil_War
		}
		effects = {
			faction_button_click = {
				if = {
					limit = {
						has_country_flag = GEO_show_factions
					}
					clr_country_flag = GEO_show_factions
				}
				else = {
					set_country_flag = GEO_show_factions
					GEO_Get_Men_Pop = yes
					GEO_Get_Trad_Pop = yes
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = GEO_show_traditionalist
						}
					}
					set_country_flag = GEO_show_menshevik
				}
			}
		}
		ai_enabled = {
			tag = GEO
			has_global_flag = Second_Russian_Civil_War
		}
		ai_test_interval = 240
		ai_test_variance = 0
		ai_check = {
			tag = GEO
			has_political_power > 39.99
			OR = {
				NOT = {
					has_idea = RUS_faction_influence_block
					has_idea = RUS_rally_block
				}
			}
		}
		ai_weights = {
			faction_button_click = {
				ai_will_do = {
					base = -1
					modifier = {
						NOT = {
							has_idea = RUS_faction_influence_block
							has_idea = RUS_rally_block
						}
						NOT = {
							has_country_flag = GEO_AI_Faction_Open
						}
						add = 100
					}
				}
			}
		}
	}
}

						ideology = neutrality

					}

					}

						ideology = nationalism

					}

					add_timed_idea = {
						idea = RUS_faction_influence_block

					}

				GEO_Get_Trad_Pop = yes

			}
