scripted_gui = {
	gen_parliament_gui = {
		context_type = player_context
		window_name = "gen_parliament_gui_container"
		parent_window_name = "autonomy_gui_window"

		dynamic_lists = {
			gen_parliament_semicircle_main_grid = {
				array = gen_parliament_main_array
				index = parliament_main_index
				change_scope = no
				entry_container = gen_parliament_semicircle_main_entry
			}
			gen_parliament_member_grid = {
				array = gen_parliament_member_array
				index = parliament_member_index
				change_scope = no
				entry_container = gen_parliament_member_entry
			}
		}

		effects = {
			gen_parliament_modify_button_control_click = {
				set_temp_variable = { gen_temp_member = v }
				set_temp_variable = { gen_temp_seats = 5 }
				gen_modify_parliament_seat = yes
			}
			gen_parliament_modify_button_shift_click = {
				set_temp_variable = { gen_temp_member = v }
				set_temp_variable = { gen_temp_seats = -5 }
				gen_modify_parliament_seat = yes
			}
		}

		properties = {
			parliament_main_icon = {
				frame = v
			}
			gen_parliament_semicircle_main_icon = {
				frame = v
				x = gen_parliament_x_array^parliament_main_index
				y = gen_parliament_y_array^parliament_main_index
			}
			gen_parliament_member_icon = {
				frame = v
			}
		}
	}
}
