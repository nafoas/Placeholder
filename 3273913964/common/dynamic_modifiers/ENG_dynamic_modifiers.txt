ENG_Parliament_Economic_Intervention = { #represents how the government takes in money and intervenes (or does not intervene) in the economy. will be capped at +/- 0.50 pp/day, except intake, which should be capped at + 1
	icon = GFX_idea_Economic_Intervention
	political_power_gain = ENG_economic_intake #Positive means more money, positive thus means you're taking in more, should always be positive
	political_power_gain = ENG_economic_spending #negative means you're spending more, should always be negative
	stability_factor = ENG_economic_stability #Positive means more stability, more government corrections = more stab, should be 1:5 with political power (+0.10 pp/ +2 stab). More taxes should decrease stability, though, -2*spending - intake
	consumer_goods_factor = ENG_economic_consumer #Positive is more consumer goods, negative is less, should be 1:5 with pp for this purpose, combined effect of 2*spending + intake
	trade_opinion_factor = ENG_economic_tariffs #positive represents more free trade, negative represents tariffs. Should be -1*1:5 with intake for this purpose
	production_speed_buildings_factor = ENG_economic_building #Positive represents more stimulus, should be at most 1:3 with pp for this purpose, based on spending
	industrial_capacity_factory = ENG_economic_industry #positive represents more subsidies/spending in industry, should be at most 1:3 with pp for this purpose, based on spending
	industrial_capacity_dockyard = ENG_economic_industry #positive represents more subsidies/spending in industry, should be at most 1:3 with pp for this purpose, based on spending
}
