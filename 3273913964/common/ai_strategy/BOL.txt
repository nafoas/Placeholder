	allowed = { 
		tag = BOL
	}
	enable = {
		has_war_with = PAR
	}

		OR = {
			has_war_with = BRA
			has_war_with = PRU
			has_war_with = CHL
			has_war_with = ARG
		}
	}

	ai_strategy = {
		type = front_unit_request

		country_trigger = {
			OR = {
				tag = BRA
				tag = PRU
				tag = CHL
				tag = ARG
			}
		}

		value = -200
	}
}
