FIN_nordic_neutrality = {
	allowed = {
		tag = FIN
	}
	enable = {
		OR = {
			country_exists = NOR
			country_exists = SWE
		}
	}

	abort = {
		OR = {
			has_war_with = NOR
			has_war_with = SWE
			is_neighbor_of = GER
		}
	}

	ai_strategy = {
		type = front_unit_request

		country_trigger = {
			OR = {
				AND = {
					tag = NOR
					NOT = { has_war_with = FROM }
				}
				AND = {
					tag = SWE
					NOT = { has_war_with = FROM }
				}
			}
		}

		value = -200
	}
}
