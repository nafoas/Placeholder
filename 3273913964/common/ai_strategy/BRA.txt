BRA_dont_frontline_the_amazonas = {
	allowed = {
		tag = BRA
	}
	enable = {
		OR = {
			is_neighbor_of = VEN
			is_neighbor_of = COL
			is_neighbor_of = PRU
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request

		country_trigger = {
			OR = {
				AND = {
					tag = VEN
					NOT = {
						has_war_with = FROM
					}
				}
				AND = {
					tag = COL
					NOT = {
						has_war_with = FROM
					}
				}
				AND = {
					tag = PRU
					NOT = {
						has_war_with = FROM
					}
				}
			}
		}

		value = -200
	}
}
