PAR_ignore_neighbours_when_chaco_war = {
	allowed = { 
		tag = PAR
	}
	enable = {
		has_war_with = BOL
	}

	abort = {
		OR = {
			has_war_with = BRA
			has_war_with = ARG
		}
	}

	ai_strategy = {
		type = front_unit_request

		country_trigger = {
			OR = {
				tag = BRA
				tag = ARG
			}
		}

		value = -200
	}
}
