		}
	}

		NOT = {
			has_war_with = SIK
			has_war_with = UTP

	ai_strategy = {
		type = front_unit_request

			OR = {
				AND = {

					NOT = { has_war_with = FROM }
				}
				AND = {
					tag = MON
					NOT = { has_war_with = FROM }
				}
				AND = {
					tag = XSM
					NOT = { has_war_with = FROM }
				}
				AND = {
					tag = MHK
					NOT = { has_war_with = FROM }
				}
				AND = {
					tag = TIB
					NOT = { has_war_with = FROM }
				}
			}
		}

		value = -200
