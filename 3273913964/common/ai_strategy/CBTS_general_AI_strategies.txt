dont_waste_pp_on_relations = { #AI always spams useless 'improve relations', especially with minors. This should make e.g SOV less keen on improving relations with Liberia, and instead focus PP on useful decisions etc. 
	enable = {
		has_political_power < 300
	}
	abort = {
		has_political_power > 301
	}
	ai_strategy = {
		type = pp_spend_priority
		id = relation
		value = -10000
	}
}
dont_waste_pp_on_generals = {
    enable = {
		has_political_power < 300
	}
	abort = {
		NOT = { has_political_power < 300 }
	}

	ai_strategy = {
		type = pp_spend_priority
		id = admiral
		value = -1000
	}

	ai_strategy = {
		type = pp_spend_priority
		id = general
		value = -1000
	}
}
