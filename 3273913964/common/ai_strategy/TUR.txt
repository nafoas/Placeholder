TUR_ignore_Iran = {
	allowed = {
		original_tag = TUR
	}
	enable = {
		country_exists = PER
	}

	abort = {
		OR = {
			has_war_with = PER
			has_wargoal_against = PER
			PER = { has_wargoal_against = TUR }
		}
	}

	ai_strategy = {
		type = front_unit_request

		country_trigger = {
			tag = PER
			NOT = { has_war_with = FROM }
		}

		value = -200
	}
}
