SAm_dont_sent_troops_there = {
	allowed = {
		OR = {
			tag = CHL
			tag = PAR
			tag = BOL
			tag = ARG
			tag = URG
			tag = DOM
			tag = HAI
			tag = CUB
			tag = MEX
			tag = GUA
			tag = HON
			tag = NIC
			tag = COS
			tag = PAN
			tag = VEN
			tag = COL
			tag = ECU
			tag = PRU	
		}
	}
	enable = {
		any_other_country = {
			capital_scope = {
				NOT = { is_on_continent = south_america }
				NOT = { is_on_continent = north_america }
			}
			has_war_with = PREV
			has_capitulated = no
		}
	}
	
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request

		country_trigger = {
			capital_scope = {
				NOT = { is_on_continent = south_america }
				NOT = { is_on_continent = north_america }
			}
		}

		value = -200
	}
}
