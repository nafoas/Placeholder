	allowed = {
		always = no #vanilla put this so idk
		OR = {
			tag = USA
			tag = ENG
		}
	}
	
	# only allow invasion on europe on every third month
	
	enable = {
		set_temp_variable = { days = global.num_days }
		subtract_from_temp_variable = { days = 705545 }
		modulo_temp_variable = { days = 90 }
		check_variable = { days < 60 }
	}
	
	abort_when_not_enabled = yes
	
	ai_strategy = {
		type = front_control

		area = mainland_europe
		
		ordertype = invasion
		execute_order = no
	}
}
