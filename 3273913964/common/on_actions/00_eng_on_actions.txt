	on_war = {
		effect = {
			if = {
				limit = {
					THIS = {
						tag = ENG
						OR = {
							has_war_with = GER
							has_war_with = DDR
							has_war_with = SOV
							has_war_with = ITA

							NOT = {

							NOT = {

							NOT = {
								has_completed_focus = ENG_Anti_Communist_Start
							}

							has_country_flag = ENG_national_government_wartime

				ENG = {
					set_variable = {
						eng_debate_root = ROOT
					}
					set_variable = {
						eng_debate_from = FROM
					}
					set_country_flag = ENG_guilty_men

		}
