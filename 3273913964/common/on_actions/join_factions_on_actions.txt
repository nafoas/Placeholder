on_actions = {
	on_offer_join_faction = {
		effect = {
			### Clears all JF flags before running
			clr_global_flag = JF_Sichuan_DXH
			clr_global_flag = JF_Sichuan_TSY
			### Sets the flags to be used in scripted loc
			if = {
				limit = {
					tag = LXI
					FROM = { tag = DXH }
				}
				set_global_flag = JF_Sichuan_DXH
			}
			else_if = {
				limit = {
					tag = LXI
					FROM = { tag = TSY }
				}
				set_global_flag = JF_Sichuan_TSY
			}
		}
	}
}
