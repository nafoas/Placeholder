	on_offer_join_faction = {
		effect = {
			FROM = { #Stay with your pal dummy
				add_ai_strategy = {
					type = alliance

					value = 200
				}
			}
			if = {
				limit = {
					NOT = {
						FROM = {
							tag = event_target:new_faction_member
						}
					}
				}
				save_global_event_target_as = faction_leader
				FROM = {
					save_global_event_target_as = new_faction_member
				}
				every_country = {
					limit = {
						is_ai = no
					}
					sound_effect = "diplomatic_notification_08"
					if = {
						limit = {
							NOT = {
								has_country_flag = JOINED_FACTION_POPUP1
							}
						}
						set_country_flag = JOINED_FACTION_POPUP1
					}
					else_if = {
						limit = {
							NOT = {
								has_country_flag = JOINED_FACTION_POPUP2
							}
						}
						set_country_flag = JOINED_FACTION_POPUP2
					}
					else_if = {
						limit = {
							NOT = {
								has_country_flag = JOINED_FACTION_POPUP3
							}
						}
						set_country_flag = JOINED_FACTION_POPUP3
					}
					else = {
						random_list = {
							50 = {
								clr_country_flag = JOINED_FACTION_POPUP1
								set_country_flag = JOINED_FACTION_POPUP1
							}
							50 = {

	on_declare_war = {
		effect = {
			if = {
				limit = {
					ROOT = { tag = GER }
					FROM = { tag = DEN }
				}
				hidden_effect = {
					# GER = {

					DEN = {
						country_event = { hours = 6 id = den.1 }
						delete_unit = {
							disband = yes
						}
					}
					if = {
						limit = {
							NOT = {
								ICE = { has_idea = neutrality_idea }
							}
						}
						ICE = { add_ideas = neutrality_idea }
					}
				}
			}
			### Germany declares war on Poland ###
			if = {
				limit = {
					ROOT = { tag = GER }
					FROM = { tag = POL }
				}
				hidden_effect = {
					log = "[GetDateText]: Germany declares war on Poland."

					POL = {
						add_ideas = POL_is_not_yet_lost
					}

					random_list = {
						90 = {
							GER = { set_country_flag = GER_has_good_luck }
							log = "[GetDateText]: Germany has Good luck with the French."
						}
						10 = {
							GER = { set_country_flag = GER_has_bad_luck }
							log = "[GetDateText]: Germany has Bad Luck with the French."
						}
					}

					GER = { news_event = { id = ger_news.10 } }

					if = {
						limit = { has_global_flag = MR_pact_flag }
						log = "[GetDateText]: Soviets get event that starts mission to invade Poland. MR pact"
						SOV = {
							country_event = {
								id = mr_pact_event.7
								days = 4
							}
						}
					}

					#Netral countries declare neutrality
					DEN = { add_ideas = neutrality_idea }

	on_capitulation = {
		effect = {
			if = { #If Denmark capitulates while being a puppet of Germany
				limit = {
					ROOT = { tag = DEN }
					NOT = { 
						FROM = { tag = GER }
					}
					DEN = { has_idea = DEN_german_occupation }
				}
				if = {
					limit = {
						DEN = { has_cosmetic_tag = DEN_GER }
					}
					DEN = { drop_cosmetic_tag = yes }
				}
				DEN = {
					remove_ideas = DEN_german_occupation
					remove_ideas = DEN_danish_neutrality
				}
			}
			if = { #If Germany capitulates and Denmark is their puppet
				limit = {
					ROOT = { tag = GER }
					DEN = { is_subject_of = GER }
				}
				if = {
					limit = {
						DEN = { has_idea = DEN_german_occupation }
					}
					DEN = { remove_ideas = DEN_german_occupation }
				}
				if = {
					limit = {
						DEN = { has_cosmetic_tag = DEN_GER }
					}
					DEN = { drop_cosmetic_tag = yes }
				}
				GER = {
					set_autonomy = {
						target = DEN
						autonomy_state = autonomy_free
						end_wars = yes
					}
				}
				DEN = {
					remove_ideas = DEN_danish_neutrality
					leave_faction = yes
					country_event = { id = den.11 } #Liberation of Denmark
				}
			}

			if = { #Italy is in faction with GER, is not a puppet and capitulates after Germany does
				limit = {
					ROOT = { tag = ITA }
					ITA = {
						is_subject = no
						is_in_faction_with = GER
					}
					GER = { has_capitulated = yes }
				}

				FROM = {
					country_event = {
						id = postwar.4
					}
				}
			}

			if = { #End of WW2. German surrender
				limit = {
					ROOT = { tag = GER }
					FROM = {
						OR = {
							tag = ALY
							is_in_faction_with = ALY
							is_subject_of = ALY
							AND = {
								tag = SOV
								NOT = { has_war_with = ALY }
							}
							AND = {
								tag = RUS
								NOT = { has_war_with = ALY }
							}
							#What if it capitulates to a country that is neither of these? TO THINK
						}
					}
				}
				FROM = {
					country_event = { id = postwar.1 hours = 20 }
				}
				
				if = { #Italy
					limit = {
						ITA = {
							is_subject_of = GER
						}
					}
					FROM = {
						country_event = {
							id = postwar.4
						}
					}
				}

				if = {
					limit = {
						OR = {
							HUN = { has_country_flag = axis_country }
							HUN = { is_in_faction_with = GER } #just in case
						}
					}
					FROM = {
						country_event = {
							id = postwar.3
							hours = 2
						}
					}
				}
			}
		}
	}
