					}
				}

			}
			else_if = {
				limit = {
					ROOT = {
						tag = POL
					}
					FROM = {
						tag = GER
					}
					has_global_flag = MR_pact_flag
					NOT = { has_global_flag = SOV_invading_Poland_MR }
					has_global_flag = SOV_get_no_land_in_POL #Soviets didnt attack and mission timed out
					NOT = {
						GER = { has_war_with = SOV }
						SOV = { has_war_with = POL }
					}
					country_exists = SOV
					country_exists = GER
				}

				set_global_flag = GER_can_form_ggv

				every_state = {
					limit = {
						has_state_flag = GER_MR
					}
					clr_state_flag = GER_MR
				}
				every_state = {
					limit = {
						has_state_flag = SOV_MR
					}
					clr_state_flag = SOV_MR
				}
			}
		}
