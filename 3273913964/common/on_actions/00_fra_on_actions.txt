	on_new_term_election = {
		effect = {
			# 1936 Election - Hopefully more complex after 0.1
			if = {
				limit = {
					has_capitulated = no
					is_puppet = no
					tag = FRA
					date < 1937.1.1
				}

				if = {
					limit = {
						has_country_flag = SFIO_radical
						has_country_flag = AD_concentration
					}
					country_event = france.302
				}
				else_if = {
					limit = {
						has_country_flag = SFIO_radical
						has_country_flag = AD_right_wing
					}
					if = {
						limit = {
							is_historical_focus_on = no
							NOT = {
							    is_ai = yes
							    OR = {
								    has_game_rule = { rule = FRA_select_path option = FRA_PF_VICTORY }
									has_game_rule = { rule = FRA_select_path option = FRA_AD_VICTORY }
								}
							}
						}
						random_list = {
							75 = { # Popular Front Wins
								country_event = france.302
							}
							25 = { # Right AD Coalition Wins
								country_event = france.303
							}
						}
					}
					else_if = {
					    limit = {
						    is_ai = yes
							has_game_rule = { rule = FRA_select_path option = FRA_PF_VICTORY }
						}
					    country_event = france.302
					}
					else_if = {
					    limit = {
						    is_ai = yes
							has_game_rule = { rule = FRA_select_path option = FRA_AD_VICTORY }
						}
					    country_event = france.303
					}
					else = {
						country_event = france.302
					}
				}
				else_if = {
					limit = {
						has_country_flag = SFIO_gauche
						OR = {
							has_country_flag = AD_right_wing
							has_country_flag = AD_concentration
						}
					}
					country_event = france.303
				}

				else_if = { #fallback trigger just in case
					limit = {
						tag = FRA
						date < 1937.1.1
					}
					country_event = france.302
				}
			}
		}
	}

	on_declare_war = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							FROM = { tag = FRA }
							ROOT = {
								is_major = yes
							}
						}
						AND = {
							ROOT = { tag = FRA }
							FROM = {
								is_major = yes
							}
						}
					}
					NOT = {
						FRA = {
							OR = {
								has_idea = all_adults_serve
								has_idea = service_by_requirement
								has_idea = scraping_the_barrel
							}
						}
					}
				}
				if = {
					limit = {
						FRA = { has_idea = FRA_The_Great_War }
					}
					FRA = { remove_ideas = FRA_The_Great_War }
				}
				FRA = {
					country_event = {
						id = france.404 #Mobilisation Generale
						days = 1
					}
				}
			}
		}
		effect = {
			if = { # Check if they are members of Bloc
				limit = {
					FROM = {
						has_country_flag = member_bloc_dor
					}
					ROOT = {

					}
				}

				custom_effect_tooltip = bloc_dor_war_tt
				disband_bloc_dor = yes
			}
		}
	}

	on_capitulation = {

				log = "[GetDateText]: on_actions/fra_on_actions/on_capitulation: France has capitulated to Germany. Armistice set up event should fire"

				every_state = {
					limit = {
						has_dynamic_modifier = {
							modifier = GER_france_is_collapsing
						}
					}
					remove_dynamic_modifier = {
						modifier = GER_france_is_collapsing
					}
				}
				
				# TODO: different Germanies (Hitler, DNVP, KPD) fire different events. Do it with ifs
				if = { #Communist Armistice
					limit = {
						FROM = { tag = DDR }

					log = "[GetDateText]: on_actions/fra_on_actions/on_capitulation: communist armistice"
					DDR = {
						country_event = { id = fra_armistice.21 }
					}
				}
				else_if = {
					limit = {
						FROM = {
							tag = GER
						}
					}
					log = "[GetDateText]: on_actions/fra_on_actions/on_capitulation: normal armistice"
					GER = {
						country_event = { id = fra_armistice.17 }
					}
				}
				if = {
					limit = {
						FRA = { is_in_faction_with = ENG }
					}
					set_victory_points = {
						province = 1145
						value = 20
					}
					set_victory_points = {
						province = 7132
						value = 10
					}
					FRA = {
						become_exiled_in = {
							target = ENG
							legitimacy = 10

				}
			}

			if = {
				limit = {
					is_puppet_of = VIC
					FROM = {
						has_war_together_with = FRA
					}
				}
				end_puppet = ROOT
